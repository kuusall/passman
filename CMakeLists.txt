cmake_minimum_required(VERSION 3.20)

# Prefer Clang when the user has not selected a compiler explicitly:
# cmake -S . -B build
# Users can still override this with -DCMAKE_CXX_COMPILER=/path/to/compiler.

if(NOT DEFINED CMAKE_CXX_COMPILER AND NOT CMAKE_CXX_COMPILER)
    find_program(PASSMAN_CLANGXX clang++)

    if(PASSMAN_CLANGXX)
        set(
            CMAKE_CXX_COMPILER
            "${PASSMAN_CLANGXX}"
            CACHE FILEPATH
            "C++ compiler used to build passman"
            FORCE
        )
    endif()
endif()

project(passman VERSION 1.0.0 LANGUAGES CXX)

include(CTest)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(
    PASSMAN_ENABLE_SANITIZERS
    "Enable AddressSanitizer and UndefinedBehaviorSanitizer"
    OFF
)


# Source files


add_executable(passman
    src/cli.cpp
    src/clipboard.cpp
    src/clipboard_timer.cpp
    src/commands.cpp
    src/crypto.cpp
    src/entry.cpp
    src/export.cpp
    src/main.cpp
    src/password.cpp
    src/session.cpp
    src/vault.cpp
)


# Include directories


target_include_directories(passman PRIVATE
    "${PROJECT_SOURCE_DIR}/include"
)


# Compiler warnings

target_compile_options(passman PRIVATE
    -Wall
    -Wextra
    -Wpedantic
    -Wconversion
    -Wshadow
)

# libsodium
find_path(
    SODIUM_INCLUDE_DIR
    sodium.h
)

find_library(
    SODIUM_LIBRARY
    sodium
)

if(NOT SODIUM_INCLUDE_DIR OR NOT SODIUM_LIBRARY)
    message(FATAL_ERROR
        "libsodium not found. Install it with: sudo pacman -S libsodium"
    )
endif()

target_include_directories(passman PRIVATE
    "${SODIUM_INCLUDE_DIR}"
)

target_link_libraries(passman PRIVATE
    "${SODIUM_LIBRARY}"
)

if(APPLE)
    target_compile_definitions(passman PRIVATE
        CLIPBOARD_COPY_CMD="pbcopy"
        CLIPBOARD_PASTE_CMD="pbpaste"
    )
else()
    target_compile_definitions(passman PRIVATE
        CLIPBOARD_COPY_CMD="wl-copy"
        CLIPBOARD_PASTE_CMD="wl-paste"
    )
endif()

include(GNUInstallDirs)
install(TARGETS passman
    RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
)

# sanitizers
if(PASSMAN_ENABLE_SANITIZERS)

    foreach(target_name
        passman
        passman_crypto_test
        passman_vault_test
        passman_entry_test
        passman_password_test
        passman_cli_test
        passman_search_test)
        if(TARGET ${target_name})
            target_compile_options(${target_name} PRIVATE
                -fsanitize=address,undefined
                -fno-omit-frame-pointer
            )
            target_link_options(${target_name} PRIVATE
                -fsanitize=address,undefined
            )
        endif()
    endforeach()

endif()
